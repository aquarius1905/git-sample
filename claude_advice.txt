```python
# app/core/config.py
from pydantic import Field, validator
from pydantic_settings import BaseSettings

class Settings(BaseSettings):
    model_config = ConfigDict(env_prefix="")
    
    s3_bucket_name: str = Field(..., min_length=3, max_length=63)
    key_prefix: str = Field(default="etl-output", regex=r"^[a-zA-Z0-9\-/]+$")
    aws_endpoint_url: Optional[str] = Field(default=None)
    
    @validator('s3_bucket_name')
    def validate_bucket_name(cls, v):
        if not v or v == "your-bucket-name":
            raise ValueError("S3バケット名は必須です")
        return v
```
